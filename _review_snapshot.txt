<script id="notif-bell-js">
(function () {
  'use strict';
  var core = window.FPU_PORTAL || {};
  var $ = function (id) { return document.getElementById(id); };
  var esc = function (s) { return core.escapeHtml ? core.escapeHtml(s) : String(s == null ? '' : s); };
  var timeAgo = function (d) { return core.timeAgo ? core.timeAgo(d) : ''; };

  var POLL_MS = 30000;
  var panel, bell, badge, list;
  var timer = null;
  var bound = false;

  function renderList(rows) {
    if (!rows.length) {
      list.innerHTML = '<div class="notif-empty">You\'re all caught up.</div>';
      return;
    }
    list.innerHTML = rows.slice(0, 10).map(function (n) {
      var unread = n.isRead === false || n.isRead === undefined;
      return '<button type="button" class="notif-item ' + (unread ? 'unread' : '') + '" data-id="' + esc(n.id) + '" data-link="' + esc(n.link || '') + '">'
        + '<div class="notif-item-top">'
        +   '<span class="notif-item-title">' + (unread ? '<span class="notif-dot"></span>' : '') + esc(n.title || 'Notification') + '</span>'
        +   '<span class="notif-item-time">' + esc(timeAgo(n.createdAt) || '') + '</span>'
        + '</div>'
        + (n.body ? '<div class="notif-item-body">' + esc(n.body) + '</div>' : '')
        + '</button>';
    }).join('');
    list.querySelectorAll('.notif-item').forEach(function (el) {
      el.addEventListener('click', function () { openItem(el); });
    });
  }

  async function openItem(el) {
    var id = el.dataset.id;
    var link = el.dataset.link || '';
    try { await core.portalFetch('/api/portal/notifications/' + id + '/read', { method: 'POST' }); }
    catch (e) { /* silent */ }
    closePanel();
    refresh();
    if (link) {
      var hash = link.split('#').pop();
      if (window.FPU_SPA && window.FPU_SPA.navigatePortal && hash && /^[a-z-]+$/.test(hash)) {
        window.FPU_SPA.navigatePortal(hash);
        return;
      }
      window.location.href = link;
    } else if (window.FPU_SPA && window.FPU_SPA.navigatePortal) {
      window.FPU_SPA.navigatePortal('portal-notifications');
    }
  }

  async function refresh() {
    try {
      var res = await core.portalFetch('/api/portal/notifications?limit=10');
      var rows = (res && res.data) || [];
      var unread = (res && res.unread) || 0;
      if (badge) {
        if (unread > 0) { badge.hidden = false; badge.textContent = unread > 99 ? '99+' : String(unread); }
        else { badge.hidden = true; }
      }
      if (list) renderList(rows);
    } catch (err) { /* silent */ }
  }

  function openPanel() {
    if (!panel) return;
    panel.hidden = false;
    if (bell) bell.setAttribute('aria-expanded', 'true');
    refresh();
  }
  function closePanel() {
    if (!panel) return;
    panel.hidden = true;
    if (bell) bell.setAttribute('aria-expanded', 'false');
  }

  async function markAllRead() {
    try {
      await core.portalFetch('/api/portal/notifications/read-all', { method: 'POST' });
      if (core.showToast) core.showToast('All marked as read.', 'success');
      refresh();
    } catch (err) { if (core.showToast) core.showToast(err.message, 'error'); }
  }

  function goToNotifications() {
    closePanel();
    if (window.FPU_SPA && window.FPU_SPA.navigatePortal) {
      window.FPU_SPA.navigatePortal('portal-notifications');
    } else {
      window.location.hash = 'portal-notifications';
    }
  }

  function bind() {
    if (bound) return;
    panel = $('notif-panel');
    bell = $('notif-bell');
    badge = $('notif-badge');
    list = $('notif-list');
    if (!panel || !bell || !list) return false;
    bound = true;

    bell.addEventListener('click', function (e) {
      e.stopPropagation();
      if (panel.hidden) openPanel(); else closePanel();
    });
    var markAll = $('notif-mark-all');
    if (markAll) markAll.addEventListener('click', function (e) { e.stopPropagation(); markAllRead(); });
    var viewAll = $('notif-view-all');
    if (viewAll) viewAll.addEventListener('click', function (e) { e.stopPropagation(); goToNotifications(); });

    document.addEventListener('click', function (e) {
      if (!panel || panel.hidden) return;
      if (panel.contains(e.target) || bell.contains(e.target)) return;
      closePanel();
    });
    document.addEventListener('keydown', function (e) {
      if (e.key === 'Escape' && panel && !panel.hidden) closePanel();
    });

    refresh();
    if (timer) clearInterval(timer);
    timer = setInterval(refresh, POLL_MS);
    window.addEventListener('hashchange', refresh);
    return true;
  }

  // Retry loop: keep trying every 100ms until the bell exists
  function init() {
    if (bind()) return;
    var attempts = 0;
    var retry = setInterval(function () {
      attempts++;
      if (bind() || attempts > 50) clearInterval(retry);
    }, 100);
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', init);
  } else {
    init();
  }
})();
</script>
